\chapter{Examples of simple dynamical systems}
\section{Important example: autonomous ODEs}
\begin{figure}[ht]
    \centering
    \includegraphics[width=0.3\textwidth]{img/cap1/cauchy.png}
    \caption{Representation of the quantities in an autonomous ODE.}
    \label{fig:aut_ode}
\end{figure}
The most obvious example of a dynamical system is given by an ODE, which takes the form of:
\begin{equation*}
    \dot{x}=f(x)
\end{equation*}
where $M\subseteq\mathbb{R}^d,\,f:M\rightarrow\mathbb{R}^d$. 

We are looking at a curve of which we know the first time derivative, and we know said derivative to be a function of position, but not of time, as shown in \autoref{fig:aut_ode}. 

The aforementioned problem has, potentially, infinite solutions. What interests us are what are called Cauchy problems or, more commonly,\footnote{This is true for the whole world, apart from Italy and France, which only call those Cauchy problems. The latter has obvious reasons to do so. Italy, on the other hand, hasn't got so many.} IVP, Initial Value Problems.

\begin{definition}[Cauchy problem]
    A Cauchy problem is given by the following linear system:
    \begin{equation}
    \begin{cases}
    \dot{x}=f(x) \\
    x(0)=x_0
    \end{cases}
    \label{eq:cauchy}
    \end{equation}
\end{definition}
Cauchy's theorem tells us that under some conditions, which we assume, the solution exists, and it's unique. (those conditions being that f(x) is a Lipschitz function). 

We shall start with the classic ODE example: the harmonic oscillator.

\begin{figure}[ht]
    \centering
    \includegraphics[width=0.3\textwidth]{img/cap1/harmonic_oscillator.png}
    \caption{Representation of a harmonic oscillator.}
    \label{fig:harmonic_oscillator}
\end{figure}

\begin{example}[Harmonic oscillator]
    The one dimensional harmonic oscillator is a physical system in which a mass is attracted to the origin by a force that depends on the distance to the origin itself, as shown in \autoref{fig:harmonic_oscillator}. This is described by:
    \begin{equation}
        \ddot{x}=-w^2x
    \end{equation}
    At first glance, it may look like it does not fit the definition of a Cauchy problem, since it is a second derivative instead of a first one. Turns out, all high order ODEs may be reduced to a single order one. 

    The trick is to rename the first derivative. in this case:
    \begin{equation*}
        \begin{cases}
            \dot{x}=v \\
            \dot{v}=-w^2x
        \end{cases}
    \end{equation*}
    Or, in matrix form:
    \begin{equation*}
        \dot{\begin{bmatrix}x \\ v\end{bmatrix}}=\begin{bmatrix} v \\ -w^2x\end{bmatrix}=f\left(\begin{bmatrix} x \\ v\end{bmatrix}\right)
    \end{equation*}
    We did increase the dimension of the phase space, but that does not get in the way of calling this a Cauchy problem.
\end{example}

Other than uniqueness and existence, we will also be implicitly assuming  all solutions to be global, i.e. defined $\forall t\in\mathbb{R}$, which is a much more restrictive condition. While finding a problem for which uniqueness and existence are not true is quite hard, the same is not true for a Cauchy problem without a global solution. Case in point:

\begin{example}[of a non global solution]
    \begin{equation*}
        \begin{cases}
            \dot{x}=x^2 \\
            x(0)=1
        \end{cases}
    \end{equation*}
    where $M=\mathbb{R}$.
    Solving it by abusing math a bit:
    \begin{equation*}
        \frac{dx}{dt}=x^2 \Leftrightarrow \frac{dx}{x^2}=dt\Leftrightarrow\int_1^x\frac{d\xi}{\xi^2}=\int_0^td\tau\Leftrightarrow\left[-\frac{1}{\xi}\right]^x_1=t  \Leftrightarrow 1-\frac{1}{x}=t\Leftrightarrow x=\frac{1}{1-t}
    \end{equation*}
    \begin{tikzpicture}
    \centering
    \begin{axis}
    % Plot left branch (x < 1)
    \addplot [domain=-3:0.9, samples=100, blue, thick] {1/(1-x)};

    % Plot right branch (x > 1)
    \addplot [domain=1.1:5, samples=100, blue, thick] {1/(1-x)};

    % Draw vertical asymptote at x = 1
    \draw [dashed, red, thick] (axis cs:1,-10) -- (axis cs:1,10);
    \end{axis}
    \end{tikzpicture}

    As you can see, it blows up asymptotically at 1, so it is not a global solution.
\end{example}

We shall now see the concept of flow.

\begin{figure}[ht]
    \centering
    \includegraphics[width=0.5\textwidth]{img/cap1/flux.png}
    \caption{Representation of flow.}
    \label{fig:flow}
\end{figure}

\begin{definition}
    If $x(t;x_0)$ is the solution of \autoref{eq:cauchy}, the flow of such system is:
    \begin{equation}
        \phi^t(x_0)=x(t;x_0)
        \label{eq:flow}
    \end{equation}
\end{definition}

\begin{proposition}
    $\phi^t_{t\in\mathbb{R}}$, as defined in \autoref{eq:flow} satisfies the group property.
\end{proposition}

\begin{proof}
    The group properties are two:
    \begin{equation}
        \phi^0=id_M
        \label{eq:identity}
    \end{equation}
    \begin{equation}
        \phi^{t+s}=\phi^s\circ \phi^t
        \label{eq:composition}
    \end{equation}

    \autoref{eq:identity} is trivial, because by definition $\phi^0(x_0)=x_0\forall\, x_0\in M$.

    The second property is a bit harder. First of all, by definition:
    \begin{equation}
        \phi^{t+s}=\phi^t\circ\phi^s\Leftrightarrow \phi^t(\phi^s(x_0))=\phi^{t+s}(x_0)\forall x_0
        \label{eq:proof}
    \end{equation}
    We define $y_1(t)=\phi^t(\phi^s(x_0))$ and $y_2(t)=\phi^{t+s}(x_0)$. We'll prove that $y_1(t)$ and $y_2(t)$ are both solutions of the following IVP:
    \begin{equation}
        \begin{cases}
            \dot{y}(t)=f(y(t)) \\ y(0)=\phi^s(x_0) \forall s\in\mathbb{R},\ x_0\in M
        \end{cases}
    \end{equation}
    Because of Cauchy's theorem of uniqueness, if both solve the IVP, they must be equal, thus the proposition will be proved because of \autoref{eq:proof}. 

    The initial condition is true by definition:
    \begin{equation*}
        \begin{matrix}
        y_1(0)=\phi^{0}(\phi^s(x_0))=\phi^s(x_0) \cr \cr y_2(0)=\phi^{0+s}(x_0)=\phi^{s}(x_0) \cr
        \end{matrix} \Leftrightarrow y_1(0)=y_2(0)=\phi^s(x_0)\forall s\in \mathbb{R}
    \end{equation*}

    As for the other condition, for $y_1$ it's true by definition, first by the definition of $y_1$ itself, then by the definition of $\phi^t(x_0)$:
        \begin{equation*}
        \begin{aligned}
        \frac{dy_1}{dt}(t)=\frac{d\phi^t}{dt}(\phi^s(x_0))=\frac{dx(t;\,\phi_s(x_0))}{dt}= \\
        f(x(t;\,\phi^s(x_0)))=f(\phi^t(\phi^s(x_0)))=f(y_1(t))
        \end{aligned}
        \end{equation*}

    
    where x is the solution of the IVP over which the flow is defined, while f() for which $\dot{x}=f(x)$ is true.

    In the same way, by employing the definitions of $y_2$ and flow, aided by the chain rule, we may find that the temporal derivative of $y_2$ is equal to $f(y_2)$, where f() is the same as the one for which that is true for $y_1$:
        \begin{equation*}
        \begin{aligned}
        \frac{dy_2}{dt}(t)=\frac{d\phi^{t+s}}{dt}(x_0)=\frac{dx(t+s;\,x_0)}{d(t+s)}\frac{d(t+s)}{dt}= \\
        f(x(t+s;\,x_0))=f(\phi^{t+s}(x_0))=f(y_2(t))
        \end{aligned}
        \end{equation*}
    
    So, $y_1$ and $y_2$ both solve the aforementioned IVP for every t, thus we proved the proposition.


\end{proof}
Let's now look a bit into the concept of flow. The flow behaves in such a way(this is why it's called a flow!) in which it never intercepts. It looks, if drawn in a stylized way, as a series of lines that never intercept, because of the uniqueness of the solution of the Cauchy problem, as shown in \autoref{fig:flow}.

\begin{figure}[ht]
    \centering
    \includegraphics[width=0.3\textwidth]{img/cap1/intersection.png}
    \caption{Representation of the hypothesis used to prove \autoref{prop:intersection}.}
    \label{fig:intersection}
\end{figure}

\begin{proposition}
    We define two flows $\phi^{t_1}(x_1)$ and $\phi^{t_2}(x_2)$. If $t_1\geq t_2$ and $x_1\neq x_2$, they shall never intercept. 
    \label{prop:intersection}
\end{proposition}
\begin{proof}
    RAA\footnote{Reductio ad absurdum. I will use this to signal that I am proving by contradiction.} Assume that we have two trajectories that are completely distinct apart from one singular interception. Let's call the singular interception $\bar{x}$. The following is true by definition:
    \begin{equation*}
        \bar{x}=\phi^{t_1}(x_1)=\phi^{t_2}(x_2)
    \end{equation*}
    But then, by definition of flow:
    \begin{equation*}
        \phi^{-t_2}(\bar{x})=\phi^{-t_2}(\phi^{t_2}(\bar{x}))=x_2
    \end{equation*}
    but at the same time:
    \begin{equation*}
        \phi^{-t_2}(\bar{x})=\phi^{-t_2}(\phi^{t_1}(x_1))=\phi^{t_1-t_2}(x_1)
    \end{equation*}
    The hypothesis here that the interception is just one. Starting from said hypothesis, we found another interception, namely $x_2$. This would mean, because of the uniqueness of the solution of the Cauchy problem, that the trajectories are exactly the same: but that is impossible, because we assumed that they were distinct apart from one singular point.
\end{proof}

Now: keep in mind that a global solution is not hard to imagine.

\begin{example}[of global solution ascertained by visual inspection]
    Suppose M is a two-dimensional domain and suppose you know that at the boundary the vector field f always points inside. And suppose you know that this vector field is smooth and defined inside all the domain. Then you are assured that the solution is global. This because the vector field cannot "escape" the domain, and it covers all of it. 
\end{example}

Usually you can find a global solution by iteratively move inside small intervals. 

Now, for a little result that is more of a curiosity than anything else.

\begin{proposition}
    If $\phi^t$, a continuous DS on $M\subseteq\mathbb{R}^d$, is smooth enough\footnote{Meaning $C^1$ Lipschitz, in this case.}, then it is the flow of this ODE:
\begin{equation*}
    \dot{x}=\left.\frac{\partial \phi^t}{\partial t}\right|_{t=0}
\end{equation*} 
\end{proposition}
\begin{proof}
    As it is a continuous DS, it has to satisfy the semi-group property. 
    
    Keeping this in mind, to verify the proposition we need to define the function
    \begin{equation*}
        x(t,\,x_0):t\rightarrow\phi^{t}(x_0)
    \end{equation*}
    for all $x_0\in M$, then verify that it satisfies the ODE, given the initial condition:
    \begin{equation*}
        x(0,\,x_0)=x_0
    \end{equation*}
    The initial condition is true by definition, because $x(0, x_0)=\phi^0(x_0)=x_0$.

    Then, we must verify that it does indeed solve the ODE. By definition:
    \begin{equation*}
        \dot{x}(t, x_0)=\frac{\partial \phi^t(x_0)}{\partial t}
    \end{equation*}
    then, by employing the chain rule\footnote{
            Not immediately obvious:
            \begin{equation*}
                \frac{\partial \phi^{t+s}(x_0)}{\partial s}=\frac{\partial \phi^{t+s}(x_0)}{\partial(t+s)}\frac{\partial (t+s)}{\partial s}=\frac{\partial \phi^{t+s}(x_0)}{\partial(t+s)}
            \end{equation*}
            Now, if we evaluate that at $s=0$:
            \begin{equation*}
                \left.\frac{\partial \phi^{t+s}(x_0)}{\partial s}\right|_{s=0}=\left.\frac{\partial \phi^{t+s}(x_0)}{\partial(t+s)}\right|_{s=0}=\frac{\partial \phi^t(x_0)}{\partial t}
            \end{equation*}
        }:
    \begin{align*}
        \frac{\partial \phi^t(x_0)}{\partial t}=
        \left.\frac{\partial \phi^{t+s}(x_0)}{\partial s}\right|_{s=0}=\left.\frac{\partial}{\partial s}\phi^{s}(\phi^t(x_0))\right|_{s=0}= \\
        \left.\frac{\partial\phi^{s}}{\partial s}(\phi^t(x_0))\right|_{s=0}=\left.\frac{\partial\phi^{s}}{\partial s}(x(t,x_0))\right|_{s=0}            
    \end{align*}
    Then, straightforwardly:
    \begin{equation*}
        \dot{x}(t, x_0)=\left.\frac{\partial\phi^{s}}{\partial s}(x(t,x_0))\right|_{s=0}=\left.\frac{\partial\phi^{t}}{\partial t}(x(t,x_0))\right|_{t=0}
    \end{equation*}
    As ugly as that may be, it is true, since s is just a name: we can change s to t without second thoughts because it is a partial derivative, which fixes $x(t, x_0)$ in place, thus avoiding uncomfortable situations with the temporal dependency of the position.
\end{proof}

\section{Non-autonomous ODEs}
\begin{definition}
    A non-autonomous ODE is a differential equation of the form:
    \begin{equation*}
        \dot{x}=f(x,\,t)
    \end{equation*}
\end{definition}
This kind of problem brings with itself some problems, mainly that you cannot use the theory of dynamical systems. Let's see why.
\begin{example}
    \begin{figure}[ht]
        \centering
        \begin{tikzpicture}
        \begin{axis}
        % Plot left branch (x < 1)
        \addplot [domain=0:2.99, samples=100, blue, thick] {0};

        % Plot right branch (x > 1)
        \addplot [domain=3.01:7, samples=100, blue, thick] {x-3};

        \end{axis}
        \end{tikzpicture}
        \caption{Plot of f(x, t) for the example.}
    \end{figure}
    
    Let's say that:
    \begin{equation*}
        f(x,\,t)=\begin{cases}
            0 \qquad 0<x<3 \\
            x-3 \qquad 3<x
        \end{cases}
    \end{equation*}

    Calling the flow $\phi^t(x_0)=x(t,\,x_0)$, we could easily see that the semi-group property is not satisfied. Say that you apply $\phi^2$ twice:
    \begin{equation*}
        \phi^2(x_0)=x_0 \Leftrightarrow \phi^2(\phi^2(x_0))=\phi^2(x_0)=x_0
    \end{equation*}
    quite trivially:
    \begin{equation*}
    \phi^4(x_0)=x_0+1 \Leftrightarrow \phi^4\neq\phi^2(\phi^2(x_0))
    \end{equation*}
\end{example}

This renders absolutely impossible to use DS theory, at first glance. How do you go around it? 


The first way one could think of is to notice that the IVP's initial condition is really not that much of a constraint. Let's see why this is not true.
\begin{proposition}
    We define an IVP as follows:
    \begin{equation*}
        \begin{cases}
            \dot{x}=f(x) \\
            x(t_0\neq0)=x_0
        \end{cases}
    \end{equation*}
    we define $\phi^{t_0, t}(x_0)=x(t, x_0, t_0)$. In this case, you lose the semi-group property, i.e. $\phi^{t, s}\circ \phi^{t_0, t}=\phi^{t_0,s}$.
\end{proposition}
\begin{proof}
    Homework.
\end{proof}

There is a correct workaround, and it is the following.

\begin{proposition}
    Suppose you have the following IVP, where $x\in M\subseteq\mathbb{R}^d$:
\begin{equation*}
    \begin{cases}
        \dot{x}=f(x, t) \\
        x(0)=x_0
    \end{cases}
\end{equation*}
Which is built on a non-autonomous ODE. We set $y=(x, t) \in M\times\mathbb{R}\subseteq\mathbb{R}^{d+1}$ and $F(y)=F(x, t)= (f(x, t), 1)$.

Then, we consider the IVP:
\begin{equation*}
    \begin{cases}
       \dot{y}=F(y)\\
       y(0)=(x_0, 0) 
    \end{cases}
    \label{eq:y_ode}
\end{equation*}
Where the dot is the derivative with respect to the newly defined "time" s,  for which the following is true:
\begin{align*}
    \begin{bmatrix}
        \frac{dx}{ds}(s) \\ \frac{dt}{ds}(s)
    \end{bmatrix}&=\begin{bmatrix}
        f(x(s), t(s)) \\ 1
    \end{bmatrix} \\
   \begin{bmatrix}
    x(0) \\ t(0)
   \end{bmatrix}&=\begin{bmatrix}
    x_0 \\ 0
   \end{bmatrix}
\end{align*}

Then this is an autonomous ODE, with respect to t. 
\end{proposition}
\begin{proof}
First things first, we can actually see how s behaves in respect to t from the bottom members of the matrices of its definition:
    \begin{equation*}
    \frac{dt}{ds}(s)=1, t(0)=0 \Leftrightarrow t(s)=s
    \end{equation*}
so s and t are actually the same. So, by looking at the top members, you find the previous problem with respect to t.
\end{proof}
\begin{observation}
    What we are doing is the classic physics trick to consider time as a dimension as the other ones.
\end{observation}

So, basically, if you have a non-autonomous ODE you can make it an autonomous one.

\section{Fixed and periodic points}
\begin{definition}
    Considering the usual $\mathbb{G}=\mathbb{R}, \mathbb{R}^+_0, \mathbb{Z}, \mathbb{N}$, for a DS ${\phi^t}_{t\in \mathbb{G}}$, the orbit or trajectory of $x\in M$ is defined as:
    \begin{equation*}
        O(x)=\{\phi^t(x)|t\in \mathbb{G}\}
    \end{equation*}
    In the cases of interest, we will use \begin{equation*}
        O^{t(-)}(x)=\{\phi^t(x)|t\geq(\leq)0\}
    \end{equation*}
    and call it future or forward (past or backward, in the case with minus and $\leq$) orbit or trajectory. 
\end{definition}
\begin{definition}
    $\bar{x}\in M$ is called a fixed point if $\phi^t(\bar{x})=\bar{x}, \forall t>0$.

    Else, $\bar{x}$ is called a periodic point if $\exists T>0|\phi^T(\bar{x})=\bar{x}$, and T is called a period for $\bar{x}$. 

    In the case of maps, the set of fixed points is sometimes denoted $Fix(T)={x\in M|T(x)=x} $.
\end{definition}
\begin{observation}
    Clearly, in periodic cases $\phi^{t+T}(\bar{x})=\phi^t(\bar{x})$.\footnote{Provable by group property on the definition.}
\end{observation}
\begin{observation}
    Notice that if $x\in Fix(T)$, then $T^n(x)=x$ which is true if and only if $x\in Fix(T^n)$; more formally:
    \begin{equation*}
        x\in Fix(T)\Rightarrow T^n(x)=x \Leftrightarrow x\in Fix(T^n)
    \end{equation*} 

    Also, one can very easily notice that a periodic point of T is by definition a point in $Fix(T^n)$ for some $n\in\mathbb{Z}$.
\end{observation}

Graphically, this means that by plotting $T(x)$ and then the bisectrix of the first and the third plane, and then finding their interceptions, you obtain the fixed points, as shown in \autoref{fig:g_f_p}.

Riffing from this simple method, one can develop a really easy graphical method to find the orbit of a 1D map, which works as follows, as shown in \autoref{fig:g_a_f_p}:
\begin{enumerate}
    \item Find $T(x_0)$ on the graph;
    \item Draw a horizontal line from $T(x_0)$ to the bisectrix of the first and third plane;
    \item From the intersection of the bisectrix and the just drawn horizontal line, draw a vertical one until intersecting T(x);
    \item Repeat.
\end{enumerate}

\subsection{Classification of fixed points}
\begin{figure}
    \centering
    \includegraphics[width=0.7\textwidth]{img/cap1/fixed_points.png}
    \caption{Representation of the types of fixed points.}
    \label{fig:fixed_points}
\end{figure}
Here, we will assume that $M$ is a topological space. 
\begin{definition}
    A topological space is one in which the notions of open sets, closed sets and neighbourhood are defined.\footnote{Iffy as hell as a definition.}
\end{definition}
\begin{definition}
    A fixed point $\bar{x}\in M$ is called:
    \begin{itemize}
        \item \textbf{stable} if for every neighbourhood $U$ of $\bar{x}$ exists a neighbourhood V\footnote{can be trivially proven to be inside U} of $\bar{x}$ such that for every $x\in V$, $\phi^t(x)$ is in $U$ for every positive $t$\footnote{This means that if you look close enough, you can look even closer and say if I start there I will stay close in the future.}. In math: \begin{equation*}
            \forall U \text{ nbhd of } \bar{x}\,\exists V \text{ nbhd of } \bar{x}|\forall x\in V, \phi^t(x)\in U,\, \forall x\geq0
        \end{equation*}
        \item \textbf{attractive} (or attracting) if it is stable and for every $x\in V$, $\phi^t(x)$ tends to $\bar{x}$ for infinite time. In math: \begin{equation*}
            \forall x\in V, \lim_{t\rightarrow+\inf}\phi^t(x)=\bar{x}
        \end{equation*}
        \item \textbf{unstable} if it is not stable, i.e.\footnote{Id est, which is Latin for "that is".} there is a neighbourhood $U$ of $\bar{x}$ such that for every neighbourhood $V$ of $\bar{x}$, exists and $x\in V$ and a $\bar{t}$ which is a function of said x $\bar{t}(x)$ such that $\phi^{\bar{t}}(x)$ is not in $U$. In math: \begin{equation*}
            \exists U \text{ nbhd of } \bar{x}|\forall V \text{ nbhd of }\bar{x},\, \exists x\in V,\, \bar{t}=\bar{t}(x)|\phi^{\bar{t}}(x)\notin U
        \end{equation*} 
        \item \textbf{repulsive} (or repelling) if there is a $U$ neighbourhood of $\bar{x}$ such that for every $x\in U$ there is a $\bar{t}$ which is a function of x $\bar{t}(x)$ such that $\phi^{\bar{t}}(x)$ is not in $U$. In math:
        \begin{equation*}
            \exists U\text{ nbhd of }\bar{x}|\forall x \in U \exists\bar{t}=\bar{t}(x)|\phi^{\bar{t}}(x)\notin U
        \end{equation*} 
    \end{itemize} 
    All of these definitions are exemplified by the drawings in \autoref{fig:fixed_points}.
\end{definition}
\subsection{Examples of 1d dynamical systems}
Let's now try to tackle some actual dynamical systems; the following examples are proposed models of population growth.
\begin{example}[by Thomas Malthus]
    In 1798, Thomas Malthus proposed the following model while trying to model the growth of population:
    \begin{equation*}
        \dot{x}(t)=\lambda x(t)\Rightarrow x(t)=x_0e^{\lambda t} (t\in\mathbb{R}, \lambda\in\mathbb{R})
    \end{equation*}
    Alternatively, with discrete time:
    \begin{equation*}
        x_{n-1}=\alpha x_n (\alpha\in\mathbb{R}^+)
    \end{equation*}
    which, using the introduced terminology, means that the map is defined as follows:
    \begin{align*}
    T\colon&\mathbb{R}^+\longrightarrow\mathbb{R}^+\\
    &\phantom{+}x\mapsto \alpha x,
     \end{align*}
    This is not hard to solve, and the solution is:
    \begin{equation*}
        x_n=x_0\alpha^n
    \end{equation*}
    Heuristically, this means that the growth of the population is proportional to the number of people, which is reasonable. 

    One quick note: oftentimes discrete time is more realistic than real time, since you cannot actually measure the population at all times.
\end{example}
Those models are rarely realistic, though. There are more advanced ones.
\begin{example}[by Pierre Francois Velhurst]
    Popularized by Robert May in 1978, the logistic model was developed by P.F. Velhurst in 1838 as means of modeling population growth.
    \begin{equation*}
        x_{n+1}=\alpha(A-x_n)x_n
    \end{equation*}
    The new modeling "step" is adding a sort of cap to the amount of people an ecosystem can support. Once that cap, which is parametrized in the equation by $A$, is surpassed, the population will have trouble surviving. This can be understood considering $\alpha(A-x_n)$ as growth coefficient: it changes in respect with the distance of the real population with the cap, also factually limiting the population: it would not make sense for $\alpha(A-x_n)$ to be negative since (differently from the continuous case) having that coefficient to be negative would make $x_{n+1}$ negative, which does not make sense. 
    
    We want to know the map of this system. We shall start by rearranging the terms and then changing variable:
    \begin{equation*}
        \frac{x_{n+1}}{A}=\alpha \left(1-\frac{x_n}{A}\right)\frac{x_n}{A}
    \end{equation*}
    Let's call $y=\frac{x}{A}$. Then:
    \begin{equation*}
        y_{n+1}=ry_n(1-y_n), \qquad r=A\alpha
    \end{equation*}
    We may now define:
    \begin{align*}
    T\colon&[0, 1]\longrightarrow[0, 1]\\
    &\phantom{+}x\mapsto \alpha rx(1-x),
     \end{align*}
    The justification for the domain to be in $[0, 1]$ is obviously that, as stated before, $0\leq x\leq A$ so $0\leq\frac{x}{A}\leq1$.

    As we may see in \autoref{fig:logistic}, the maximum occurs when $x=\frac{1}{2}$, which means that the maximum is $T(\frac{1}{2})=\frac{r}{4}$. Since we want it to be a self-map(domain and codomain are the same space), we have the added constraint $r\in[0, 4]$. 

    We are now interested in finding the fixed points of this map. Luckily, this is one of the rare cases in which we may find the exact solutions, which we will call $x_t$ and $x_f$\footnote{For trivial and fixed.}:
    \begin{equation*}
        T(x)=x\Leftrightarrow -rx^2+rx=x\Leftrightarrow-rx\left(x-\frac{r-1}{r}\right)=0\Rightarrow\begin{cases}
            x_t=0 \\ x_f=\frac{r-1}{r}
        \end{cases}
    \end{equation*}

    Note that $x_f$ only "exists" for $r\geq1$, since we are in [0,1]. 
    
    We had a hunch, when we found the graphical method for the orbit, that the nature of the fixed point was related in some way with the value of the derivative of the map, more precisely with it being more or less than 1. We will prove this in general in the future, but for the time being, let's just calculate the derivative of this map:
    \begin{equation*}
        T'(x)=-2rx+r\Rightarrow\begin{cases}
            T'(x_t)=r \\
            T'(x_f)=-2r\frac{r-1}{r}+r=2-r
        \end{cases}
    \end{equation*}
    Which, in turn, implies that:
    \begin{equation*}
        |T'(0)|<1\Leftrightarrow r<1\text{ and }|T'(x_f)|<1\Leftrightarrow1<r<3
    \end{equation*} 
    So, we may expect different behaviors in those four different regions of the domain of $r\in[0,4]$, namely $[0,1]$, $[1,3]$, $[3, 4]$. Let's see how true that is.
    
    (algoritmo grafico sui cosi per i vari casi) 
    
    \begin{itemize}
        \item \textbf{$r\leq1$}: when r is less than 1, the only fixed point is given by $x_t=0$. In this case, the orbit converges quite quickly towards the origin;
        \item \textbf{$1<r\leq2$}: when $r$ is between 1 and 2, the maximum is below the bisectrix (apart from $r=2$), which  implies that $x_f$ is in the first half of the map, or $x_f\leq0.5$. In this case, all the trajectories converge to $x_f$, apart from the one that already starts at $x_t=0$;
        \item \textbf{$2<r\leq3$}: when $r$ increases (without surpassing 3) the behaviour is the same: the orbit converges towards $x_f$ if $x_0\neq0$ but in a different way, namely it oscillates around $x_f$ until it reaches it, overshooting and undershooting it until finally converging;
        \item \textbf{$3<r\lessapprox3.45$}: increasing further, we can see that the orbit becomes periodic, oscillating between two points with $T=2$. So, it may be interesting to see the behaviour of the squared map, where we can see that there are 4 periodic points (namely $x_t$, $x_f$ and the two points between which it oscillates). We may also notice that the trajectory there tends to one of the two periodic points (if you don't start either in $x_t$ or $x_f$), which is a matter of "scope": the squared map only sees one of the two points in the periodic motion, thus those are attractive periodic points for the orbit as seen by $T^2(x)$;
        \item \textbf{$r\gtrapprox3.45$}: once you go over circa 3.45, the movement becomes much more unpredictable: it is still periodic, but the periodicity will increase a lot even with small variations of r, and as such the motion will becomes much more chaotic;

\subsection{Characterization of fixed points for 1D maps}
We shall now demonstrate the hunch we had about the correlation between the derivative and the nature of the fixed point, at least in 1D.
\begin{proposition}
    Let $T:I\rightarrow I$, I connected set\footnote{Meaning either an interval, half line, or $\mathbb{R}.$} in $\mathbb{R}$, $T\in C^1(I)$. Let $\bar{x}\in Fix(T)$. Then:
    \begin{enumerate}
        \item if $|T'(\bar{x})|<1$ then $\bar{x}$ is attractive;
        \item if $|T'(\bar{x})|>1$ then $\bar{x}$ is repelling; 
    \end{enumerate}
    [disegni del diagramma di cobweb]
\end{proposition}
\begin{proof}[1]
Since $T\in C^1$, the derivative is continuous, so there exists a neighbourhood $U=(\bar{x}-\delta, \bar{x}+\delta)$\footnote{$]\bar{x}-\delta, \bar{x}+\delta[$} such that for every $ x\in U, |T'(x)|\leq\lambda<1$. In math:
\begin{equation*}
    T\in C^1, T'(\bar{x})<1\Rightarrow\exists U \textbf{ nbhd of } \bar{x}|\forall x\in U,\, |T'(x)|\leq\lambda<1,\, \lambda \text{ is close to 1.}
\end{equation*}

Let $x_0\in U$ be an arbitrary point in the neighbourhood, and we shall call $x_n=T^n(x_0)$.

We can notice that, if for all $n\geq1$, $x_n$ is in U and $|x_n-\bar{x}|\leq\lambda^n|x_0-\bar{x}|$, or:
\begin{equation}
    n\geq1\Rightarrow x_n\in U \land |x_n-\bar{x}|\leq\lambda^n|x_0-\bar{x}| 
    \label{eq:claim}
\end{equation} 
then $x_n\rightarrow x_0$, which would prove that the point is attractive. Thus, to prove the proposition we need to prove \autoref{eq:claim}.

This can be done by induction.
First, let' say that $n=1$. In that case, by definition:
\begin{equation*}
    |x_1-\bar{x}|=|T(x_0)-T(\bar{x})|
\end{equation*}
by using Lagrange's mean value theorem\footnote{Let $f \colon [a, b] \to \mathbb{R}$ be $f\in C^1$.  Then $\exists \, \psi \in (a, b)$ such that:
\[
f'(\psi) = \frac{f(b) - f(a)}{b - a}
\]} we can find that, for some $\psi\in(x_0, \bar{x})\in U$\begin{equation*}
    |T(x_0)-T(\bar{x})|=|T'(\phi)(x_0-\bar{x})|\leq\lambda|x_0-\bar{x}|
\end{equation*}
which is a part of the claim. Also, by cancelling the passages:
\begin{equation*}
    |x_1-\bar{x}|\leq\lambda|x_0-\bar{x}|\Rightarrow x_1\in U
\end{equation*}
which completes the verification for n=1.

We are left to prove that if the statement is true for n, then it is true for n+1. The process up to a point is exactly the same: \begin{equation*}
    |x_{n+1}-\bar{x}|=|T(x_n)-T(\bar{x})|=|T'(\psi_n)(x_n-\bar{x})|\leq\lambda|x_n-\bar{x}|
\end{equation*}
Now, we can find by iteration that $\lambda|x_n-\bar{x}|\leq \lambda^{n+1}|x_0-\bar{x}|$, which what we were trying to demonstrate. $x_{n+1}\in U$ is trivially proved in the same way as the first part. 
\end{proof}
\begin{proof}[2]
    Homework.
\end{proof}
\begin{definition}
    If $|T'(\bar{x})|=1$ the fixed point is called neutral or indifferent.
\end{definition}
\subsection{Fixed points for linear flows in more than 1D}
Say you have a linear ODE with constant data:
\begin{equation}
    \begin{cases}
        \dot{x}=Ax, \qquad x\in\mathbb{R}^d, \, \text{A is a constant matrix}\in\mathbb{R}^{d\times d} \\
        x(0)=x_0\in\mathbb{R}^d
    \end{cases}
    \label{eq:ivp_2d}
\end{equation}
The solution of the initial value problem is:
\begin{equation*}
    x(t)=e^{tA}x_0, \, \forall A\in\mathbb{R}^{d\times d}
\end{equation*}
Where the exponential is defined as follows.
\begin{definition}
    If B is a matrix, then $e^B=\sum_{k=0}^{\infty}\frac{B^k}{k!}$.
    Note that the series converges in the space of $d\times d$ matrices with respect to any norm, because if $||B||<\infty\Rightarrow||B^k||\leq ||B||^k$. \footnote{The definition might sound a bit iffy, in the sense that the whole "in any norm" thing is a big leap of faith. Actually, in finite dimensions, if $|\,|,||\,||, |||\,|||$ are all norms of V, $dim(V)<\infty$, then there exists a $C>1$ such that\begin{equation*}
        C^{-1}|v|\leq||v||\leq C||v||
    \end{equation*}
    so, if you prove a limit with some norm it is actually true for all of them due to the definition of norm.} 
\end{definition}
\begin{observation}
    There is no guarantee that A is diagonalizable. One could point out that you can prove that finding non-diagonalizable matrices is quite hard, and even then they are infinitely "close" to the diagonalizable ones, but we do not care about the space of matrices per se, we care about single matrices individually. So we have to consider the possibility that our matrix is not diagonalizable.
\end{observation}

Luckily, linear algebra comes to our aid, thanks to Jordan\footnote{Six FMVPs, six scoring titles, five MVPs, one DPOY, 10 times All-NBA, nine All-defensive team, three ASGMVP. Greatest PPG in the history of the NBA, both in the regular season and in the playoffs, and an important theorem to his name. Lebron ain't got no theorem to his name.}'s theorem.
\begin{theorem}[Jordan's theorem]
    Any matrix $A\in\mathbb{R}^{d\times d}$ can be put in Jordan form:
    \begin{equation*}
        \begin{bmatrix}
        J_1 & 0 & \cdots & 0 \\
        0 & J_2 & \cdots & 0 \\
        \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & \cdots & J_k
        \end{bmatrix}
    \end{equation*}
    where a Jordan block\footnote{https://youtu.be/tRhYR5DB2rg?si=qMxcfBx-5ecyhcjr} $J_i$ takes the form: \begin{equation*}
        J_i = \begin{bmatrix}
        \lambda_i & 1 & 0 & \cdots & 0 \\
        0 & \lambda_i & 1 & \cdots & 0 \\
        0 & 0 & \lambda_i & \cdots & 0 \\
        \vdots & \vdots & \vdots & \ddots & 1 \\
        0 & 0 & 0 & \cdots & \lambda_i
        \end{bmatrix}
    \end{equation*}
    Note that Jordan blocks can be of different sizes.

    Also, there exists a matrix $C\in\mathbb{C}^{d\times d}$ with a non-zero determinant, such that $A=CA_JC^{-1}$. C is called the base change matrix.
\end{theorem}

For now on, we will always assume $A$ is already in Jordan form. This is rapidly justified.

\begin{proposition}
    Starting from the problem given by \autoref{eq:ivp_2d}, one may find an equivalent problem with the Jordan form of $A$.
\end{proposition}
\begin{proof}
    Starting from Jordan's theorem:
    \begin{equation*}
        A=CA_JC^{-1}\Leftrightarrow A_J=C^{-1}AC
    \end{equation*}
    which implies:
    \begin{equation*}
        \dot{x}=Ax\Leftrightarrow\dot{x}=CA_JC^{-1}x\Leftrightarrow C^{-1}\dot{x}=A_JC^{-1}x
    \end{equation*}
    So, by simply renaming $y=C^{-1}x$ and so $y_0=C^{-1}x_0$ we may easily find:\begin{equation*}
        \begin{cases}
            y=A_Jy \\
            y(0)=y_0
        \end{cases}
    \end{equation*}
    which is an equivalent problem.
\end{proof}

\begin{example}
\label{ex:complex_coeff}
    Let the A in \autoref{eq:ivp_2d} be:
    \begin{equation*}
        A=\begin{bmatrix}
            a+ib & 0 \\
            0 & a-ib
        \end{bmatrix},\qquad a,\,b\in\mathbb{R} 
    \end{equation*}
    This implies, based on what was stated at the beginning of the section:
    \begin{equation*}
        e^{tA}=\begin{bmatrix}
            e^{at}e^{ibt} & 0 \\
            0 & e^{at}e^{-ibt}
        \end{bmatrix}
    \end{equation*}
    since the initial condition is real, we may infer that the solution will be real. We will nonetheless prove it. 
    
    Let's do another change of variable, in order to have a real matrix.

    Consider the following matrix:
    \begin{equation*}
        D=\frac{1}{\sqrt{2}}\begin{bmatrix}
            1 & i \\
            i & 1
        \end{bmatrix} \Leftrightarrow D^{-1}=\frac{1}{\sqrt{2}}\begin{bmatrix}
            1 & -i \\
            -i & 1
        \end{bmatrix} 
    \end{equation*}
    We shall use $z=D^{-1}y$. We can get to do so by using the fact that $DD^{-1}=D^{-1}D=I$ on $\dot{y}=Ay$ as following:
    \begin{equation*}
        \dot{y}=D^{-1}DADD^{-1}y \Leftrightarrow D^{-1}\dot{y}=D^{-1}ADD^{-1}y\Leftrightarrow \dot{z}=Bz
    \end{equation*} 
    where $B=D^{-1}AD$. 
    
    We are trying to find a real equation, so we need to prove that
    $B\in\mathbb{R}^{2\times 2}$ and $e^{tB}\in\mathbb{R}^{2\times 2}$. 

    First of all, $e^{tB}\in\mathbb{R}^{2\times 2}$ implies $B\in\mathbb{R}^{2\times 2}$ because:
    \begin{equation*}
    \left.\frac{d}{dt}e^{tB}\right|_{t=0}=B
    \end{equation*}\footnote{This can be proved, if you are not convinced:
    \begin{equation*}
        \frac{de^{tA}}{dt}=\frac{d}{dt}\sum_{k=0}^{\infty}\frac{t^kA^k}{k!}=\sum_{k=0}^{\infty}\frac{d}{dt}\frac{t^kA^k}{k!}
    \end{equation*}
    But $\frac{d}{dt}\frac{t^0A^0}{0!}=0$, so we may exclude the term with $k=0$, obtaining the following:
    \begin{equation*}
        \sum_{k=0}^{\infty}\frac{d}{dt}\frac{t^kA^k}{k!}=\sum_{k=1}^{\infty}\frac{d}{dt}\frac{t^kA^k}{k!}=\sum_{k=1}^{\infty}\frac{t^{k-1}A^{k-1}}{(k-1)!}A=A\sum_{j=0}^{\infty}\frac{t^{j}A^{j}}{(j)!}=Ae^{tA}
    \end{equation*}
    which also converges with $||\,||$ in $\mathbb{R}^{d\times d}$.

    It is actually as a consequence of this that $r(t)=e^{tA}r_0$ is the solution of the following IVP:\begin{equation*}
        \begin{cases}
            \dot{r}=Ar \\
            r(0)=r_0
        \end{cases}
    \end{equation*}
    In fact, $r(0)=Ir_0=r_0$ and \begin{equation*}
        \dot{r}(t)=Ae^{tA}r_0=Ar(t)
    \end{equation*}
    }

    So we only need to prove $e^{tB}\in\mathbb{R}^{2\times 2}$. We shall start by observing that:\begin{equation*}
        e^{tB}=e^{tD^{-1}AD}=\sum_{k}\frac{t^k}{k!}(D^{-1}AD)^{k}=D^{-1}e^{tA}D
    \end{equation*}
    because $(D^{-1}AD)^{k}=D^{-1}A^kD$ since the $D$s simplify themselves\footnote{Imagine $k=2$. Then you have $D^{-1}ADD^{-1}AD=D^{-1}AAD$.}. We can then insert the actual values of $D$, $D^{-1}$ and $e^{tA}$ in that, and then compute the value:
    \begin{align*}
        D^{-1}e^{tA}D=\frac{1}{\sqrt{2}}\begin{bmatrix}
            1 & -i \\
            -i & 1
        \end{bmatrix}\begin{bmatrix}
            e^{at}e^{ibt} & 0 \\
            0 & e^{at}e^{-ibt}
        \end{bmatrix}\frac{1}{\sqrt{2}}\begin{bmatrix}
            1 & i \\
            i & 1
        \end{bmatrix}= \\
        \frac{e^{at}}{2}\begin{bmatrix}
            e^{ibt} & -ie^{-ibt} \\
            -ie^{ibt} & e^{-ibt}
        \end{bmatrix}\begin{bmatrix}
            1 & i \\
            i & 1
        \end{bmatrix}=\frac{e^{at}}{2}\begin{bmatrix}
            e^{ibt}+e^{-ibt} & ie^{ibt}-ie^{-ibt} \\
            -ie^{ibt}+ie^{-ibt} & e^{ibt}+e^{-ibt}
        \end{bmatrix}
    \end{align*}
    Which, remembering the polar form for complex numbers:
    \begin{equation*}
        \frac{e^{at}}{2}\begin{bmatrix}
            e^{ibt}+e^{-ibt} & ie^{ibt}-ie^{-ibt} \\
            -ie^{ibt}+ie^{-ibt} & e^{ibt}+e^{-ibt}
        \end{bmatrix}=e^{at}\begin{bmatrix}
            cos(bt) & -sin(bt) \\
            sin(bt) & cos(bt)
        \end{bmatrix}
    \end{equation*}
    So we have proved that\begin{equation*}
        D^{-1}e^{tA}D=e^{at}\begin{bmatrix}
            cos(bt) & -sin(bt) \\
            sin(bt) & cos(bt)
        \end{bmatrix}
    \end{equation*}
    which is not only a real matrix, which is what we wanted to prove, but is also the propagator, thus we have solved the IVP. The solution is, indeed, $z(t)=e^{tB}z_0$. Alternatively:\begin{equation*}
        \begin{bmatrix}
            x \\ y
        \end{bmatrix}=e^{at}R_{bt}\begin{bmatrix}
            x_0 \\ y_0
        \end{bmatrix}
    \end{equation*}
    where $R_{bt}$ is a rotation matrix indicating a rotation of magnitude bt, which is the one we found. This makes it especially easy to visualize the drawings, shown in \autoref{fig:spiral_phase_portraits}.

\section{Linearized ODEs around a fixed point}
Suppose one has the following IVP: \begin{equation*}
    \begin{cases}
        \dot{x}=f(x),\qquad x\in M\subseteq\mathbb{R}^d \\
        x(0)=x_0
    \end{cases}
\end{equation*}
where $\bar{x}$ is a fixed point. Note that:
\begin{equation*}
    \bar{x}\text{ is a fixed point}\Leftrightarrow f(\bar{x})=0\Leftrightarrow (\phi^t(\bar{x})=\bar{x}\forall t\Leftrightarrow x(t, \bar{x})\text{ is a solution})\footnote{The last one meaning that if you start from $\bar{x}$ you stay in $\bar{x}.$}
\end{equation*}

We assume that $f\in C^1$. This implies that if $x$ is close to $\bar{x}$, we can use the differential and also the jacobian matrix to approximate as follows: \begin{equation*}
    x\text{ and }\bar{x}\text{ are close}\Rightarrow f(x)=f(\bar{x})+Df_{\bar{x}}(x-\bar{x})+o(|x-\bar{x}|)\approx0+Jf_{\bar{x}}\cdot (x-\bar{x})\footnote{(as a column vector)}
\end{equation*}
where the Jacobian is defined as:\begin{equation*}
        J_f(x) = \frac{\partial f}{\partial x} =
\begin{bmatrix}
\dfrac{\partial f_1}{\partial x_1} & \dfrac{\partial f_1}{\partial x_2} & \cdots & \dfrac{\partial f_1}{\partial x_n} \\[2ex]
\dfrac{\partial f_2}{\partial x_1} & \dfrac{\partial f_2}{\partial x_2} & \cdots & \dfrac{\partial f_2}{\partial x_n} \\[2ex]
\vdots & \vdots & \ddots & \vdots \\[1ex]
\dfrac{\partial f_m}{\partial x_1} & \dfrac{\partial f_m}{\partial x_2} & \cdots & \dfrac{\partial f_m}{\partial x_n}
\end{bmatrix} \end{equation*}

By doing so, we may linearize the ODE as follows: \begin{equation*}
    \dot{x}=J_f(x-\bar{x})\Rightarrow \dot{y}=J_fy,\qquad y=x-\bar{x}
\end{equation*}

We have to remember that we are dealing with an approximated solution, extremely close to $\bar{x}$. We will later see that, in some cases, the linear system and the non-linear system are conjugate to each other for any time, if we keep close to the fixed point. To do that, we must first introduce hyperbolic points.

\begin{definition}
    The fixed point $\bar{x}$ is called hyperbolic if $D_{\bar{x}}f$ has no eigenvalue on the imaginary line, meaning that, letting $\lambda$ be an eigenvalue, $Im \lambda=0$ if $Re \lambda=0$.
\end{definition}

In 2D, hyperbolic behaviour is the one we found in \autoref{ex:hyperbolic}: one of the directions is contracting, while the other one is expanding. To be precise, situations in which the directions are both contracting (or expanding?) but with different magnitudes, as we found in the same example, are still considered hyperbolic. Because of this, we define truly hyperbolic points.\footnote{The professor made this distinction, but it is very uncommon in the literature.}

\begin{definition}
    The fixed point $\bar{x}$ is called truly hyperbolic if $D_{\bar{x}}f$ has one eigenvalue with a positive real part and one with a negative real part.
\end{definition}

To properly understand what is NOT hyperbolic, you may look at \autoref{ex:complex_coeff}, which gave us a spiral. As a homework, consider $A=[i 0 i 0]\Rightarrow B=[0 -1 1 0]$ and verify that $e^{tB}$ is not hyperbolic. 

We are now ready to explain how legitimate the linearization we did at the start of the section is.

\begin{theorem}[Hartman-Grobmann theorem for flows]
    Let $\phi^t$ be the "partial flow"\footnote{Meaning not defined for every t} of $\dot{x}=f(x)$ where \begin{equation*}
        f(x): \Omega\rightarrow\mathbb{R}^d,\, \Omega\subseteq \mathbb{R}^{d},\, f(x)\in C^1
    \end{equation*}
    Let $\bar{x}$ be a hyperbolic fixed point and let \begin{equation*}
        \tilde{\phi}^t(y)=e^{tA}y,\qquad A=Jf_{\bar{x}}
    \end{equation*} be the flow of $\dot{y}=Ay$.
    
    Then, there are a neighbourhood U of $\bar{x}$ and a neighbourhood V of 0 and a homeomorphism\footnote{A continuous bijective function with continuous inverse} h from U to V such that $\phi^t(x)=(h^{-1}\circ \tilde{\phi}^t\circ h)(x)$ for all $x\in U$ and a $t\in\mathbb{R}$ such that $\phi^t(x)\in U$, which in turn implies (it's actually an iff\footnote{If and only if}) that $h(x)$ is in U and $\tilde{\phi^t}(h(x))$ is in V.

    In mathematical terms: \begin{align*}
        \phi^t\text{ is the partial flow of }\dot{x}=f(x),\,f(x): \Omega\rightarrow\mathbb{R}^d,\, \Omega\subseteq \mathbb{R}^{d},\, f(x)\in C^1, \\
        \bar{x}\text{ is a hyberbolic fixed point},\, \tilde{\phi}^t(y)=e^{tA}y,\, A=Jf_{\bar{x}}\text{ is the flow of }\dot{y}=Ay \Rightarrow \\
        \exists U \text{ nbhd of } \bar{x},\, V \text{ nbhd of } 0,\, h: U\rightarrow V \text{ homeomorfism } | \\
        \phi^t(x)=(h^{-1}\circ \tilde{\phi}^t\circ h)(x) \forall x\in U \land \exists t \in\mathbb{R}|\phi^{t}\in U(\Leftrightarrow h(x)\in V, \tilde{\phi^t}(h(x))\in V)
    \end{align*}
\end{theorem}

\subsubsection{Application: Lotka-Volterra equation}
Now, we shall see an application of the theory, the Lotka-Volterra model for animal populations. The model is the following:
\begin{equation*}
    \begin{cases}
        \dot{x}=x(\alpha-\beta y) \\
        \dot{y}=-y(\gamma-\delta x) \qquad \alpha, \beta, \gamma, \delta >0 \\
        x(0)=x_0 \\
        y(0)=y_0 \qquad x,\, y \in \Omega,\, \Omega\subseteq (\mathbb{R}^+)^2
    \end{cases}
\end{equation*}
where x is a population of preys while y is a population of predators.

Let's look for the fixed points. The vector field is given by $f(x, y)=\left(\alpha x- \beta xy, -\gamma y+\delta xy\right)$, so we may proceed as follows: \begin{equation*}
    \begin{cases}
        x(\alpha -\beta y)=0 \\
        -y(\gamma-\delta x)=0 
    \end{cases} \Rightarrow (x_0,y_0)=(0, 0),\, (x_1, y_1)=\left(\frac{\gamma}{\delta}, \frac{\alpha}{\beta}\right)
\end{equation*}

Let's now linearize around the fixed points. We shall start by finding the Jacobian: \begin{equation*}
    Df_{x,y}=Jf_{x,y}=\begin{bmatrix}
        \partial_x f_x & \partial_y f_x \\
        \partial_x f_y & \partial_y f_y
    \end{bmatrix}=\begin{bmatrix}
        \alpha-\beta y & -\beta x \\
        \delta y & -\gamma+\delta x
    \end{bmatrix}
\end{equation*}
And then we compute it at the fixed points:
\begin{equation*}
    Jf_{x_0, y_0}=\begin{bmatrix}
        \alpha & 0 \\
        0 & -\gamma
    \end{bmatrix}
\end{equation*}
\begin{equation*}
    Jf_{x_1, y_1}=\begin{bmatrix}
        0 & -\beta\frac{\gamma}{\delta}\\
        \delta \frac{\alpha}{\beta} & 0
    \end{bmatrix}=A
\end{equation*}

We may ask ourselves if this is a hyperbolic point. That can be shown by computing the eigenvalues:
\begin{equation*}
    \det(A-\lambda I)=\lambda^2+\alpha\gamma=0 \Rightarrow \lambda= \pm i\sqrt{\alpha\gamma}
\end{equation*}
So, $x_1,\,y_1$ is not hyperbolic.

\section{Linear maps}
Let $T(x)$ be a linear map defined as follows\footnote{Assume B to be in Jordan form.}: \begin{equation}
    T(x)=Bx,\, x\in\mathbb{R}^d,\, B\in\mathbb{R}^{d\times d}
    \label{eq:lin_map}
\end{equation}
The system solves itself. The orbit O is the following\footnote{$O(x)$ may be occasionally called $x_n=B^{n}x_0$ from now on.}:
\begin{equation*}
    O(x)=(B^nx)_{n\in\mathbb{N}}
\end{equation*}

The set of fixed points of this map is given by: \begin{equation*}
    Fix(T)=Ker(B-I)
\end{equation*}
which directly implies that $0\in Fix(T)$.
In other words, finding the fixed points means to solve the following equation:
\begin{equation*}
    Bx=x \Rightarrow (B-I)x=0
\end{equation*}

           \alpha & 0 \\ 0 & \beta 
        \end{bmatrix},\qquad \alpha,\,\beta\in\mathbb{R}
    \end{equation*}

        \begin{bmatrix}
            x_n \\ y_n
        \end{bmatrix}=\begin{bmatrix}
            \alpha^n x_0 \\ \beta^n y_0
        \end{bmatrix}
    \end{equation*}

    To find the line, we want to eliminate $n$ from the system. We may easily find: \begin{equation*}
        \frac{\log\left(\left|\frac{x_n}{x_0}\right|\right)}{\log(|\alpha|)}=n=\frac{\log\left(\left|\frac{y_n}{y_0}\right|\right)}{\log(|\beta|)}
    \end{equation*}

        \log\left(\left|\frac{y_n}{y_0}\right|\right)=\frac{\log|\beta|}{\log|\alpha|}\log\left(\left|\frac{x_n}{x_0}\right|\right)&\Rightarrow\left|\frac{y_n}{y_0}\right|=\left|\frac{x_n}{x_0}\right|^{\frac{\log|\beta|}{\log|\alpha|}}\Rightarrow \\
        |y|=C|x|^{\frac{\log|\beta|}{\log|\alpha|}},\qquad& C=\left|\frac{y_0}{x_0}\right|
    \end{align*}

Also, this whole hyperbolic behaviour can explain why we observe chaos: as much as that may be pub talk, the idea is easily inferrable from what we know. Imagine you are trying to repeat a trajectory: you mill, almost surely, miss the exact initial conditions. This will be mitigated on contracting directions, in which the trajectory will get closer. But say one direction is expanding: then the new trajectory, in that direction, will exponentially get further away from the points of the initial trajectory, thus making every one of them wildly different over some minor difference in initial conditions, as shown in the picture below. Obviously, in the real world, those differences may be minimal but will nonetheless be unpredictable. In general, truly hyperbolic points are important in physics because expanding directions are really common in Hamiltonian dynamics.

    Let the B in \autoref{eq:lin_map} be defined as follows:
    \begin{equation*}
        B=\begin{bmatrix}
            \alpha & 1 \\ 0 & \alpha
        \end{bmatrix}, \qquad\alpha\in\mathbb{R}
    \end{equation*}
    \begin{equation*}
        B^n=\begin{bmatrix}
            \alpha^n & n\alpha^{n-1} \\ 0 & \alpha^n
        \end{bmatrix}\Rightarrow\begin{bmatrix}
            x_n \\ y_n
        \end{bmatrix}=\begin{bmatrix}
            x_0\alpha^n+y_0n\alpha^{n-1} \\ y_0\alpha^n
        \end{bmatrix}
    \end{equation*}
    Assuming $\alpha\in\mathbb{R}^+/\{1\}$\footnote{Homework: do the case $\alpha\in\mathbb{R}^-/{-1}$.}. We use the usual technique to find the invariant curve: \begin{equation*}
        n=\frac{\log\left(\frac{y_n}{y_0}\right)}{\log(\alpha)}
    \end{equation*}
    Plug that in $x_n$ and you get:
    \begin{equation*}
        x_n=\frac{y_n}{y_0}x_0+\frac{\log\left(\frac{y_n}{y_0}\right)}{\log(\alpha)}\frac{1}{\alpha}\frac{y_n}{y_0}\Rightarrow x=\frac{x_0}{y_0}y+\frac{y\log\left(\frac{y_n}{y_0}\right)}{\alpha \log(\alpha)}
    \end{equation*}
    (disegno dell'aspetto per alpha maggiore di 1 e anche minore ma maggiore di 0, stessa cosa che per le ODE comunque) 

    In general, the equation is given by:\begin{equation*}
        x=y(C+\frac{\log|y|}{\alpha \log|\alpha|})
    \end{equation*}

Suppose you have a map $T:M\rightarrow M$, $M\subseteq\mathbb{R}^d$ and that $T$ is $C^1$, and suppose you have a fixed point $\bar{x}\in Fix(T)$.

For x close to $\bar{x}$, $T(x)=T(\bar{x})+J_{\bar{x}}T(x-\bar{x})+o(|x-\bar{x}|)$, calling the Jacobian B we can say:\begin{equation*}
    T(x)-\bar{x}\approx B(x-\bar{x})
\end{equation*}

If we change coordinates, calling $y=x-\bar{x}$, then \begin{equation*}
    \hat{T}(y)=T(y+\bar{x})-\bar{x}
\end{equation*}
and we can say \begin{equation*}
    \hat{T}(y)\approx By
\end{equation*}
